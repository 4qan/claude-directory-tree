    **Create `client/src/components/Toast.tsx`:**

    Build a lightweight toast system using React context (no library).

    ```typescript
    import { createContext, useContext, useState, useCallback, useRef, useEffect } from 'react';
    import { cn } from '@/lib/utils';
    import { X } from 'lucide-react';

    type ToastVariant = 'success' | 'error';

    interface Toast {
      id: number;
      message: string;
      variant: ToastVariant;
      duration: number; // ms
    }

    interface ToastContextValue {
      addToast: (message: string, variant?: ToastVariant) => void;
    }

    const ToastContext = createContext<ToastContextValue>({ addToast: () => {} });

    export function useToast() { return useContext(ToastContext); }

    export function ToastProvider({ children }: { children: React.ReactNode }) {
      const [toasts, setToasts] = useState<Toast[]>([]);
      const nextId = useRef(0);

      const addToast = useCallback((message: string, variant: ToastVariant = 'success') => {
        const id = nextId.current++;
        const duration = variant === 'error' ? 6000 : 3000;
        setToasts((prev) => [...prev, { id, message, variant, duration }]);
      }, []);

      const removeToast = useCallback((id: number) => {
        setToasts((prev) => prev.filter((t) => t.id !== id));
      }, []);

      return (
        <ToastContext.Provider value={{ addToast }}>
          {children}
          <div className="fixed bottom-4 left-1/2 -translate-x-1/2 z-50 flex flex-col-reverse gap-2 pointer-events-none">
            {toasts.map((toast) => (
              <ToastItem key={toast.id} toast={toast} onDismiss={() => removeToast(toast.id)} />
            ))}
          </div>
        </ToastContext.Provider>
      );
    }

    function ToastItem({ toast, onDismiss }: { toast: Toast; onDismiss: () => void }) {
      useEffect(() => {
        const timer = setTimeout(onDismiss, toast.duration);
        return () => clearTimeout(timer);
      }, [toast.duration, onDismiss]);

      return (
        <div className={cn(
          "pointer-events-auto flex items-center gap-2 px-4 py-2 rounded-md shadow-lg text-sm min-h-[40px]",
          toast.variant === 'success' && "bg-foreground text-background",
          toast.variant === 'error' && "bg-destructive text-destructive-foreground",
        )}>
          <span className="flex-1">{toast.message}</span>
          <button onClick={onDismiss} className="shrink-0 opacity-70 hover:opacity-100" aria-label="Dismiss">
            <X size={14} />
          </button>
        </div>
      );
    }
    ```

    ```typescript
    interface ContextMenuProps {
    ```

    Rendering logic:
    - If `target.nodeKind === 'leaf'`: show full menu (Open in Editor, separator, Copy to..., Move to..., separator, Promote to Global OR Demote to Project depending on scope)
    - If `target.nodeKind === 'scope'`: show "Open Folder" only
    - If `target.nodeKind === 'category'` or `target.nodeKind === 'root'`: don't render

    Menu items:
    - Each item is a `<button>` with `text-sm h-8 px-3 w-full text-left hover:bg-muted rounded-sm flex items-center gap-2`
    - Separators: `<div className="h-px bg-border my-1" />`
    - "Copy to..." and "Move to..." show a FlyoutSubmenu on hover (onMouseEnter) listing scopes (exclude current artifact's scope)
    - "Promote to Global" shown when artifact.scope === 'project'; disabled (opacity-50 cursor-not-allowed) when artifact.type is 'hook' or 'mcp-config', with title tooltip "Hooks are managed in settings.json" / "MCP configs require manual scope editing"
    - "Demote to Project" shown when artifact.scope === 'global'; shows flyout with project scopes, same disabled rule for hook/mcp-config

    Positioning:
    - Absolute positioned at (x, y) from contextmenu event

    Client-side `typeToDir` helper (same map as server):
    ```typescript
    const TYPE_DIR_MAP: Record<string, string> = {
    ```

    For plugin artifacts, use `path.dirname(artifact.absolutePath)` as source (but since client doesn't have `path`, use `artifact.absolutePath.substring(0, artifact.absolutePath.lastIndexOf('/'))` — actually, the server handles this. Just pass `artifact.absolutePath` and let server derive plugin dir).

    Wait, the server copy route takes `sourcePath` and `destinationDir`. For plugins, the server needs to know to copy the directory. The server `copy` route already uses `fs.cp(sourcePath, destPath, { recursive: true })` where `destPath = path.join(destinationDir, path.basename(sourcePath))`. For a plugin file like `plugins/name/plugin.json`, this would copy just the file. The server needs the plugin directory path, not the file path.

    Solution: On the client, for `artifact.type === 'plugin'`, set sourcePath to the parent directory: `artifact.absolutePath.replace(/\/[^/]+$/, '')` (strip last path segment). This way `fs.cp` copies the whole directory.
