export async function createServer(targetDir: string) {
  const server = Fastify({ logger: false }).withTypeProvider<ZodTypeProvider>();
  server.setValidatorCompiler(validatorCompiler);
  server.setSerializerCompiler(serializerCompiler);
  if (process.env.NODE_ENV === 'development') {
    await server.register(cors, { origin: true });
  }
  await scanRoutes(server, targetDir);
  await registerStatic(server);
  return server;
}
```

export async function operationsRoutes(server: FastifyInstance, targetDir: string): Promise<void>
```

Import from `../../shared/operationTypes.js`: all schemas, ARTIFACT_TYPE_DIR_MAP, BLOCKED_TYPES, WARNING_MESSAGES, DIRECTORY_TYPES.

**Six POST routes:**

1. **POST /api/operations/open** -- `await open(req.body.path)`, return `{ success: true }`.

2. **POST /api/operations/copy**:
   - If `BLOCKED_TYPES.includes(artifactType)` return 400: `{ success: false, error: "This artifact cannot be copied (managed within a JSON config file)" }`.
   - Determine copy source: if `DIRECTORY_TYPES.includes(artifactType)`, use `path.dirname(sourcePath)`. Otherwise use `sourcePath`.
   - Dest: `path.join(destinationDir, path.basename(copySource))`.
   - Conflict: if `!overwrite`, try `fs.access(destPath)` -- if exists return `{ success: false, conflict: true }`.
   - Pre-flight scan: call `scanReferences(copySource)` for @-include warnings.
   - Type warnings: if `WARNING_MESSAGES[artifactType]`, add `{ type: artifactType === 'claude-md' ? 'semantic' : 'state', message: WARNING_MESSAGES[artifactType] }`.
   - Execute: `fs.mkdir(path.dirname(destPath), { recursive: true })` then `fs.cp(copySource, destPath, { recursive: true })`.
   - Return `{ success: true, destPath, warnings }`.

3. **POST /api/operations/move** -- Same as copy, plus after success: `fs.rm(copySource, { recursive: true, force: true })`. Never use `fs.rename`.

4. **POST /api/operations/promote**:
   - Guard: blocked types return 400.
   - Dest: `path.join(os.homedir(), '.claude', ARTIFACT_TYPE_DIR_MAP[artifactType], path.basename(copySource))`.
   - Same flow as move (copy + delete source).

5. **POST /api/operations/demote**:
   - Guard: blocked types return 400.
   - Dest: `path.join(targetProjectDir, '.claude', ARTIFACT_TYPE_DIR_MAP[artifactType], path.basename(copySource))`.
   - Same flow as move.

6. **POST /api/operations/describe**:
   - Read file with `fs.readFile(path, 'utf-8')`.
   - Parse with `gray-matter`. If `data.description` exists, return `{ description: String(data.description) }`.
   - Else extract first non-empty paragraph from `content.trim().split(/\n\n+/)`.
   - Else return `{ description: null }`.

**Helper `scanReferences(sourcePath: string): Promise<PreflightWarning[]>`:**
- If sourcePath is directory, find all `.md` files recursively using `fs.readdir(sourcePath, { recursive: true })`.
- If file, scan just that file.
- For each .md file, read content, find lines matching `/^@\//m` (absolute @-includes).
- For each match, create `{ type: 'reference', message: 'References external file via @-include: {matched_line.trim()}' }`.
- Deduplicate and return.
