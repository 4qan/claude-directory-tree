  targetDir: string

  );
}

export async function createServer(targetDir: string) {
  const server = Fastify({ logger: false }).withTypeProvider<ZodTypeProvider>();
  server.setValidatorCompiler(validatorCompiler);
  server.setSerializerCompiler(serializerCompiler);
  if (process.env.NODE_ENV === 'development') {
    await server.register(cors, { origin: true });

  await scanRoutes(server, targetDir);
  await registerStatic(server);
  return server;

    **Create `src/server/routes/operations.ts`:**

    Export `async function operationsRoutes(server: FastifyInstance)`.

    Define Zod schemas:
    - `OpenRequestSchema`: `{ path: z.string() }`
    - `OpenResultSchema`: `{ success: z.boolean() }`
    - `CopyRequestSchema`: `{ sourcePath: z.string(), destinationDir: z.string(), overwrite: z.boolean().default(false) }`
    - `MoveRequestSchema`: same as CopyRequestSchema
    - `OperationResultSchema`: `{ success: z.boolean(), destPath: z.string().optional(), conflict: z.boolean().optional() }`
    - `PromoteRequestSchema`: `{ sourcePath: z.string(), artifactType: z.string() }`
    - `DemoteRequestSchema`: `{ sourcePath: z.string(), artifactType: z.string(), projectClaudeDir: z.string() }`
    - `DescribeResultSchema`: `{ description: z.string().nullable() }`

    **Virtual artifact guard** (reuse in every mutation endpoint):
    ```typescript
    function assertRealPath(p: string) {
      if (p.includes('#')) throw { statusCode: 400, message: 'Virtual artifacts (hooks, MCP configs) cannot be moved as files' };
    }
    ```

    **Endpoints:**

    1. `POST /api/operations/open` — call `assertRealPath(path)`, then `await open(req.body.path)`, return `{ success: true }`.

    2. `POST /api/operations/open-folder` — `await open(req.body.path)`, return `{ success: true }`. No assertRealPath needed (scope directories are always real paths).

    3. `POST /api/operations/copy` — `assertRealPath(sourcePath)`. Compute `destPath = path.join(destinationDir, path.basename(sourcePath))`. If not overwrite: check `fs.access(destPath)`, if exists return `{ success: false, conflict: true }`. Otherwise: `await fs.mkdir(destinationDir, { recursive: true })`, `await fs.cp(sourcePath, destPath, { recursive: true })`, return `{ success: true, destPath }`.

    4. `POST /api/operations/move` — `assertRealPath(sourcePath)`. Same conflict check as copy. Then: `await fs.cp(sourcePath, destPath, { recursive: true })`, `await fs.rm(sourcePath, { recursive: true, force: true })`, return `{ success: true, destPath }`. NEVER use `fs.rename` (cross-device EXDEV risk per research).

    5. `POST /api/operations/promote` — `assertRealPath(sourcePath)`. Derive globalDir: `path.join(os.homedir(), '.claude', typeToDir(artifactType))`. typeToDir maps: command->'commands', agent->'agents', skill->'skills', memory->'memory', plan->'plans', claude-md->'' (root .claude/), plugin->'plugins'. For plugin: source dir is `path.dirname(sourcePath)`. Copy to globalDir. Return `{ success: true, destPath }`.

    6. `POST /api/operations/demote` — `assertRealPath(sourcePath)`. Derive projectDir: `path.join(projectClaudeDir, typeToDir(artifactType))`. Copy to projectDir, then `fs.rm(sourcePath, { recursive: true, force: true })`. Return `{ success: true, destPath }`.

    7. `GET /api/operations/describe?path=...` — Read file with `fs.readFile(path, 'utf-8')`. Parse with `gray-matter`. If `data.description` exists, return it. Else extract first non-empty paragraph from `content.trim().split(/\n\n+/)[0]?.trim()`. Return `{ description: result || null }`. Catch errors (file not found) return `{ description: null }`.

    **Helper `typeToDir`:**
    ```typescript
    const TYPE_DIR_MAP: Record<string, string> = {
      command: 'commands', agent: 'agents', skill: 'skills',
      memory: 'memory', plan: 'plans', plugin: 'plugins',
      'claude-md': '', // CLAUDE.md lives at .claude/ root
    };
    function typeToDir(artifactType: string): string {
      return TYPE_DIR_MAP[artifactType] ?? '';
    }
    ```
