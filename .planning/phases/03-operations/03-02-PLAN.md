    **Create `client/src/components/ArtifactDetailPanel.tsx`:**

    ```typescript
    import { ICON_MAP, TYPE_LABELS } from '@/components/tree/iconMap';
    import type { Artifact } from '@/lib/types';

    interface ArtifactDetailPanelProps {
      artifact: Artifact | null;
    }

    function getDescription(artifact: Artifact): string {
      if (artifact.frontmatter?.description) {
        return String(artifact.frontmatter.description);
      }
      return 'No description available.';
    }

    export function ArtifactDetailPanel({ artifact }: ArtifactDetailPanelProps) {
      if (!artifact) return null;

      const Icon = ICON_MAP[artifact.type] ?? ICON_MAP['unknown'];
      const typeLabel = TYPE_LABELS[artifact.type] ?? artifact.type;

      return (
        <aside className="w-[280px] shrink-0 border-l border-border bg-muted/50 overflow-y-auto transition-all duration-150">
          <div className="p-4 space-y-3">
            <h2 className="text-base font-semibold text-foreground truncate">{artifact.name}</h2>

            <div className="flex items-center gap-2 text-sm text-muted-foreground">
              <Icon size={16} className="shrink-0" />
              <span>{typeLabel}</span>
            </div>

            <div className="text-xs text-muted-foreground">
              {artifact.scope === 'global' ? 'Global' : artifact.projectId}
            </div>

            <div className="text-xs text-muted-foreground font-mono truncate" title={artifact.absolutePath}>
              {artifact.absolutePath}
            </div>

            <div className="border-t border-border pt-3">
              <p className="text-sm text-foreground">{getDescription(artifact)}</p>
            </div>
          </div>
        </aside>
      );
    }
    ```
