    3. Add a `onKeyDown` handler on the tree container div (merge with headless-tree's existing onKeyDown from getContainerProps). The approach: get the existing props from `tree.getContainerProps()`, wrap its onKeyDown:

    ```typescript
    const containerProps = tree.getContainerProps();
    const handleTreeKeyDown = (e: React.KeyboardEvent) => {
      // Let headless-tree handle arrow keys, Enter, etc. first
      containerProps.onKeyDown?.(e);

      // Cmd+C: copy selected artifact to clipboard
      if ((e.metaKey || e.ctrlKey) && e.key === 'c') {
        if (!treeContainerRef.current?.contains(document.activeElement)) return;
        if (selectedIds.length === 1) {
          const data = items[selectedIds[0]];
          if (data?.nodeKind === 'leaf') {
            setClipboardArtifact(data as Artifact);
            // Don't prevent default if no artifact selected (allow normal browser copy)
            e.preventDefault();
          }
        }
      }

      // Cmd+V: paste clipboard artifact into focused scope
      if ((e.metaKey || e.ctrlKey) && e.key === 'v') {
        if (!clipboardArtifact) return;
        e.preventDefault();
        // Find the scope of the currently focused/selected item
        const targetScope = findScopeForItem(selectedIds[0]);
        if (targetScope) {
          executeOperation({
            type: 'copy',
            artifact: clipboardArtifact,
            destinationScope: targetScope,
          });
        }
      }

      // Shift+F10: open context menu on focused item
      if (e.shiftKey && e.key === 'F10') {
        e.preventDefault();
        const focusedElement = treeContainerRef.current?.querySelector('[data-focused="true"]')
          || treeContainerRef.current?.querySelector('[aria-selected="true"]');
        if (selectedIds.length === 1) {
          const data = items[selectedIds[0]];
          if (data) {
            const rect = focusedElement?.getBoundingClientRect();
            setContextMenu({
              open: true,
              x: rect ? rect.right : 100,
              y: rect ? rect.top : 100,
              target: data,
            });
          }
        }
      }

      // Cmd+F: jump to search
      if ((e.metaKey || e.ctrlKey) && e.key === 'f') {
        e.preventDefault();
        toolbarRef.current?.focusSearch();
      }

      // Escape: close context menu or return focus to tree
      if (e.key === 'Escape') {
        if (contextMenu.open) {
          setContextMenu(prev => ({ ...prev, open: false }));
        }
      }
    };
    ```

    4. Add `findScopeForItem` helper that walks up the tree item hierarchy to find the enclosing ScopeNode. Strategy: for a selected item ID, check if `items[id]` is a scope. If not, look through all scopes to find which one contains this artifact (by checking if artifact.id exists in scope.artifacts recursively).

    Simpler approach: store a map of `artifactId -> scopeId` during `buildItemMaps`. Add this to the return value. Then `findScopeForItem` is a simple lookup.

